/**
 * SoftmaxOffset — INFO 7375 Week 1 explainer, "[1000, 1000] → [0.5, 0.5]".
 * One component, six modes, for the shared-offset concept:
 *
 *   pipeline       SCORES → PROBABILITIES → PICK, revealed one box per step (no numbers)
 *   table          logits | printed probability, two columns
 *   shift          left column counts 1 → 1001 … while the right column stays frozen
 *   compare        two runs' printed digits side by side, then "bitwise identical?  True"
 *   cancel         constructed word diagram: same factor on top and bottom cancels
 *   intermediates  both inputs become the same list after subtracting the max
 *   rows           stacked input → output rows appearing on cue; optional error line
 *
 * Every number is passed in as a STRING copied from the evidence files
 * (w1-evidence/*.txt) — this component never computes a displayed value, except
 * the whole-number offset in `shift`, whose every frame is covered by the
 * 0..1000 sweep in intermediates-output.txt.
 * Timing props are seconds from the start of the beat's own narration clip.
 */
import React from 'react';
import {AbsoluteFill, interpolate, spring, useCurrentFrame, useVideoConfig, Easing} from 'remotion';
import {z} from 'zod';
import {CLAUDE, CLAUDE_FONT} from '../tokens/claude';

const rowSchema = z.object({
  input: z.string(),
  output: z.string(),
  appearAtS: z.number().default(0),
  outAtS: z.number().optional(),           // output half appears later (e.g. on "Fifty-fifty")
  highlight: z.boolean().default(false),
  dim: z.boolean().default(false),
});

export const softmaxOffsetSchema = z.object({
  mode: z.enum(['pipeline', 'table', 'shift', 'compare', 'cancel', 'intermediates', 'rows', 'card']),
  // card
  lines: z.array(z.object({text: z.string(), atS: z.number().default(0), strong: z.boolean().default(false)})).default([]),
  light: z.boolean().default(false),
  bodyScale: z.number().default(1),      // enlarge body lines on short cards
  durationS: z.number().default(15),     // copied from the beat's measured narration by sync_cues.py
  eyebrow: z.string().default('SOFTMAX · THE SHARED OFFSET'),
  title: z.string().default(''),
  source: z.string().default(''),
  label: z.string().default(''),          // "recorded output" / "constructed diagram"
  // pipeline
  step: z.number().default(1),
  stepAtS: z.number().default(0.3),      // when this step's new box / highlight appears
  captionAtS: z.number().default(1.6),   // loop caption (step 3) or question (step 4)
  // table / shift
  logits: z.array(z.number()).default([]),
  probs: z.array(z.string()).default([]),
  pcts: z.array(z.string()).default([]),
  probsAtS: z.number().default(0),
  sourceAtS: z.number().default(-1),
  offsetTo: z.number().default(1000),
  countStartS: z.number().default(0.8),
  countDurS: z.number().default(4.5),
  // compare / intermediates
  runs: z.array(z.object({label: z.string(), values: z.array(z.string())})).default([]),
  verdict: z.string().default(''),
  verdictAtS: z.number().default(0),
  sweepAtS: z.number().default(-1),
  // rows
  rows: z.array(rowSchema).default([]),
  errorInput: z.string().default(''),
  errorOutput: z.string().default(''),
  errorAtS: z.number().default(-1),
  footer: z.string().default(''),
  footerAtS: z.number().default(0),
});
export type SoftmaxOffsetProps = z.infer<typeof softmaxOffsetSchema>;

const SERIF = CLAUDE_FONT.serif;
const SANS = CLAUDE_FONT.ui;
const MONO = CLAUDE_FONT.mono;
// accent for text and lines: #A84A28 on cream = 5.42:1 (WCAG AA needs 4.5:1)
const ACC = '#A84A28';
const clamp01 = (v: number) => Math.min(1, Math.max(0, v));

const useIn = (atS: number) => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  if (atS < 0) return 0;        // hidden in this beat
  if (atS === 0) return 1;      // already on screen (carried over from the previous beat)
  return clamp01(spring({frame: frame - Math.round(atS * fps), fps,
    config: {damping: 26, stiffness: 140, mass: 0.8}}));
};

const Chrome: React.FC<{eyebrow: string; title: string; source: string; label: string;
  sourceEmph?: number; children: React.ReactNode}> =
({eyebrow, title, source, label, sourceEmph = 0, children}) => {
  const headIn = useIn(0);
  return (
    <AbsoluteFill style={{background: CLAUDE.PAGE, color: CLAUDE.INK}}>
      <div style={{position: 'absolute', top: 64, left: 108, fontFamily: SANS, fontSize: 32,
        fontWeight: 700, letterSpacing: 4, color: CLAUDE.INK_SOFT, opacity: headIn}}>{eyebrow}</div>
      {label && <div style={{position: 'absolute', top: 60, right: 108, fontFamily: SANS,
        fontSize: 30, fontWeight: 600, color: CLAUDE.INK_SOFT, border: `2px solid ${CLAUDE.BORDER}`,
        borderRadius: 999, padding: '6px 20px', opacity: headIn}}>{label}</div>}
      {title && <div style={{position: 'absolute', top: 128, left: 108, right: 108, fontFamily: SERIF,
        fontSize: 60, fontWeight: 700, lineHeight: 1.1, opacity: headIn,
        transform: `translateY(${(1 - headIn) * 12}px)`}}>{title}</div>}
      {children}
      {source && <div style={{position: 'absolute', bottom: 64, left: 108, right: 108,
        fontFamily: MONO, fontSize: 30, whiteSpace: 'nowrap',
        textDecoration: sourceEmph > 0.5 ? 'underline' : 'none',
        color: sourceEmph > 0.5 ? ACC : CLAUDE.INK_SOFT}}>{source}</div>}
    </AbsoluteFill>
  );
};

// ── pipeline ────────────────────────────────────────────────────────────────
const BOXES = [
  {h: 'SCORES', d: 'one raw score (a logit) per possible next token'},
  {h: 'PROBABILITIES', d: 'softmax converts scores so they sum to one'},
  {h: 'PICK', d: 'a weighted random draw chooses one token'},
];
const Pipeline: React.FC<SoftmaxOffsetProps> = (p) => {
  const shown = Math.min(3, p.step);
  // only the box introduced by THIS step animates in; earlier boxes are already on screen
  const ins = [useIn(p.step === 1 ? p.stepAtS : -1), useIn(p.step === 2 ? p.stepAtS : -1),
    useIn(p.step === 3 ? p.stepAtS : -1)];
  const focus = useIn(p.step >= 4 ? p.stepAtS : -1);
  const loopIn = useIn(p.step === 3 ? p.captionAtS : -1);
  const askIn = useIn(p.step >= 4 ? p.captionAtS : -1);
  const vis = (i: number) => (i >= shown ? 1 : p.step === i + 1 ? ins[i] : 1);
  return (
    <Chrome {...p}>
      <div style={{position: 'absolute', top: 340, left: 104, right: 104, display: 'flex',
        justifyContent: 'space-between', alignItems: 'center'}}>
        {BOXES.map((b, i) => {
          const o = vis(i);
          const ghost = i >= shown;
          const hot = i === 1 && focus > 0;
          const faded = p.step >= 4 && i !== 1;
          return (
            <React.Fragment key={b.h}>
              {i > 0 && <div style={{fontFamily: SANS, fontSize: 64, color: ACC,
                opacity: ghost ? 0.25 : vis(i)}}>→</div>}
              <div style={{width: 440, minHeight: 320, background: CLAUDE.CARD, borderRadius: 18,
                border: `${hot ? 5 : 2}px ${ghost ? 'dashed' : 'solid'} ${hot ? ACC : ghost ? CLAUDE.GHOST : CLAUDE.BORDER}`,
                padding: '34px 30px', boxShadow: '0 6px 24px rgba(61,57,41,0.08)',
                opacity: ghost ? 0.35 : (0.35 + 0.65 * o) * (faded ? 0.4 : 1), transform: `translateY(${(1 - o) * 24}px) scale(${1 + 0.04 * (hot ? focus : 0)})`}}>
                <div style={{fontFamily: SANS, fontSize: 40, fontWeight: 800, letterSpacing: 3,
                  color: hot ? ACC : CLAUDE.INK}}>{b.h}</div>
                <div style={{fontFamily: SERIF, fontSize: 34, lineHeight: 1.3, marginTop: 18,
                  color: CLAUDE.INK_SOFT}}>{b.d}</div>
              </div>
            </React.Fragment>
          );
        })}
      </div>
      <div style={{position: 'absolute', top: 720, left: 0, right: 0, textAlign: 'center',
        fontFamily: SERIF, fontStyle: 'italic', fontSize: 40, color: CLAUDE.INK_SOFT,
        opacity: p.step === 3 ? loopIn : 0}}>…then repeat for the next token. That's the loop.</div>
      <div style={{position: 'absolute', top: 740, left: 0, right: 0, textAlign: 'center',
        fontFamily: SERIF, fontSize: 52, color: CLAUDE.INK, opacity: askIn}}>
        What does this step keep, and what does it throw away?</div>
    </Chrome>
  );
};

// ── table + shift ───────────────────────────────────────────────────────────
const TwoColumn: React.FC<SoftmaxOffsetProps & {offset: number; frozen: number; probsIn: number}> =
({logits, probs, pcts, offset, frozen, probsIn, mode, offsetTo}) => (
  <div style={{position: 'absolute', top: 250, left: 160, right: 160}}>
    <div style={{display: 'flex', fontFamily: SANS, fontSize: 32, fontWeight: 700,
      letterSpacing: 3, color: CLAUDE.INK_SOFT, paddingBottom: 18,
      borderBottom: `3px solid ${CLAUDE.INK}`}}>
      <div style={{width: 520}}>SCORE (LOGIT)</div>
      <div>PRINTED PROBABILITY</div>
    </div>
    {logits.map((l, i) => (
      <div key={i} style={{display: 'flex', alignItems: 'baseline', padding: '26px 0',
        borderBottom: `2px solid ${CLAUDE.BORDER}`}}>
        <div style={{width: 520, fontFamily: MONO, fontSize: 84, fontWeight: 700,
          color: offset > 0 ? ACC : CLAUDE.INK, fontVariantNumeric: 'tabular-nums'}}>
          {l + offset}</div>
        <div style={{opacity: probsIn, flex: 1, borderRadius: 12, padding: '4px 18px', marginLeft: -18,
          outline: frozen > 0 ? `4px solid rgba(61,57,41,${0.55 * frozen})` : 'none'}}>
          <div style={{fontFamily: MONO, fontSize: 58, fontVariantNumeric: 'tabular-nums'}}>{probs[i]}</div>
          <div style={{fontFamily: SERIF, fontSize: 32, color: CLAUDE.INK_SOFT}}>≈ {pcts[i]}</div>
        </div>
      </div>
    ))}
    {mode === 'shift' && (
      <div style={{marginTop: 34, display: 'flex', fontFamily: SANS, fontSize: 36, fontWeight: 700}}>
        <div style={{width: 520, color: ACC}}>+ {offset} added to every score</div>
        <div style={{color: CLAUDE.INK, opacity: frozen}}>
          {offset >= offsetTo ? 'right column: never changed' : 'right column: not moving'}</div>
      </div>
    )}
  </div>
);

const Table: React.FC<SoftmaxOffsetProps> = (p) => {
  const probsIn = useIn(p.probsAtS);
  const srcIn = useIn(p.sourceAtS);
  return (
    <Chrome {...p} sourceEmph={srcIn}>
      <TwoColumn {...p} offset={0} frozen={0} probsIn={probsIn} />
    </Chrome>
  );
};

const Shift: React.FC<SoftmaxOffsetProps> = (p) => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  const t = frame / fps;
  const raw = interpolate(t, [p.countStartS, p.countStartS + p.countDurS], [0, p.offsetTo],
    {extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.inOut(Easing.cubic)});
  const offset = Math.round(raw);       // whole numbers only — each one is in the sweep evidence
  const frozen = clamp01((t - p.countStartS) / 0.6);
  const sweepIn = useIn(p.sweepAtS);
  return (
    <Chrome {...p}>
      <TwoColumn {...p} offset={offset} frozen={frozen} probsIn={1} />
      <div style={{position: 'absolute', bottom: 100, left: 160, right: 160, fontFamily: SERIF,
        fontSize: 32, color: CLAUDE.INK_SOFT, opacity: sweepIn}}>{p.footer}</div>
    </Chrome>
  );
};

// ── compare ─────────────────────────────────────────────────────────────────
const Compare: React.FC<SoftmaxOffsetProps> = (p) => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  const verdictIn = useIn(p.verdictAtS);
  const rowIn = [useIn(0.2), useIn(0.9)];
  // a highlight sweeps left→right across the digits of both rows before the verdict
  const sweep = interpolate(frame / fps, [1.6, Math.max(1.7, p.verdictAtS - 0.2)], [0, 1],
    {extrapolateLeft: 'clamp', extrapolateRight: 'clamp'});
  const n = Math.max(...p.runs.flatMap(r => r.values.map(v => v.length)), 1);
  return (
    <Chrome {...p}>
      <div style={{position: 'absolute', top: 250, left: 96, right: 96}}>
        {p.runs.map((r, ri) => (
          <div key={ri} style={{opacity: rowIn[ri] ?? 1, marginBottom: 44}}>
            <div style={{fontFamily: MONO, fontSize: 44, fontWeight: 700, marginBottom: 14,
              color: ri === 1 ? ACC : CLAUDE.INK}}>{r.label}</div>
            <div style={{display: 'flex', gap: 46}}>
              {r.values.map((v, vi) => (
                <div key={vi} style={{fontFamily: MONO, fontSize: 40, background: CLAUDE.CARD,
                  border: `2px solid ${CLAUDE.BORDER}`, borderRadius: 10, padding: '14px 18px'}}>
                  {v.split('').map((ch, ci) => (
                    <span key={ci} style={{background: ci / n < sweep ? 'rgba(217,119,87,0.22)' : 'transparent'}}>{ch}</span>
                  ))}
                </div>
              ))}
            </div>
          </div>
        ))}
      </div>
      <div style={{position: 'absolute', top: 760, left: 96, fontFamily: MONO, fontSize: 72,
        fontWeight: 700, color: CLAUDE.INK, opacity: verdictIn,
        transform: `scale(${0.92 + 0.08 * verdictIn})`, transformOrigin: 'left center',
        background: 'rgba(217,119,87,0.14)', border: `4px solid ${ACC}`, borderRadius: 14,
        padding: '10px 30px', whiteSpace: 'pre'}}>{p.verdict}</div>
    </Chrome>
  );
};

// ── cancel (constructed diagram, words only) ────────────────────────────────
const Cancel: React.FC<SoftmaxOffsetProps> = (p) => {
  const fracIn = useIn(0.2);
  const factorIn = useIn(p.probsAtS);
  const strikeIn = useIn(p.verdictAtS);
  const Chip: React.FC = () => (
    <span style={{position: 'relative', marginLeft: 26, fontFamily: SANS, fontSize: 42,
      fontWeight: 700, color: ACC, border: `3px solid ${ACC}`, borderRadius: 10,
      padding: '2px 16px', opacity: factorIn, display: 'inline-block'}}>
      × same factor
      <span style={{position: 'absolute', left: -8, right: -8, top: '50%', height: 6,
        background: CLAUDE.INK, transform: `scaleX(${strikeIn})`, transformOrigin: 'left'}} />
    </span>
  );
  return (
    <Chrome {...p}>
      <div style={{position: 'absolute', top: 270, left: 0, right: 0, display: 'flex',
        justifyContent: 'center', alignItems: 'center', gap: 50, opacity: fracIn}}>
        <div style={{fontFamily: SERIF, fontSize: 52}}>probability of a token&nbsp; =</div>
        <div style={{display: 'flex', flexDirection: 'column', alignItems: 'center'}}>
          <div style={{fontFamily: SERIF, fontSize: 52, padding: '0 20px 16px'}}>its weight<Chip /></div>
          <div style={{alignSelf: 'stretch', height: 5, background: CLAUDE.INK}} />
          <div style={{fontFamily: SERIF, fontSize: 52, padding: '16px 20px 0'}}>total of all weights<Chip /></div>
        </div>
      </div>
      <div style={{position: 'absolute', top: 580, left: 120, right: 120, textAlign: 'center',
        fontFamily: SERIF, fontSize: 44, lineHeight: 1.35, color: CLAUDE.INK_SOFT, opacity: factorIn}}>
        weight = exp(score). Add the same amount to every score,<br />and every weight grows by the same factor.</div>
      <div style={{position: 'absolute', top: 770, left: 0, right: 0, textAlign: 'center',
        fontFamily: SERIF, fontSize: 56, fontWeight: 700, opacity: strikeIn}}>
        Top and bottom both scale, so the fraction doesn't change.</div>
    </Chrome>
  );
};

// ── intermediates ───────────────────────────────────────────────────────────
const Intermediates: React.FC<SoftmaxOffsetProps> = (p) => {
  const rIn = [useIn(0.2), useIn(0.8)];
  const arrowIn = useIn(p.probsAtS);
  const verdictIn = useIn(p.verdictAtS);
  return (
    <Chrome {...p}>
      <div style={{position: 'absolute', top: 280, left: 96, right: 96}}>
        {p.runs.map((r, ri) => (
          <div key={ri} style={{display: 'flex', alignItems: 'center', marginBottom: 60,
            opacity: rIn[ri] ?? 1, fontFamily: MONO, fontSize: 56, fontWeight: 700}}>
            <div style={{width: 720, color: ri === 1 ? ACC : CLAUDE.INK}}>{r.label}</div>
            <div style={{width: 340, fontFamily: SANS, fontSize: 38, color: CLAUDE.INK_SOFT,
              opacity: arrowIn}}>minus max →</div>
            <div style={{opacity: arrowIn, background: 'rgba(217,119,87,0.14)',
              border: `4px solid ${ACC}`, borderRadius: 12, padding: '4px 22px'}}>{r.values[0]}</div>
          </div>
        ))}
      </div>
      <div style={{position: 'absolute', top: 640, left: 96, right: 96, fontFamily: SERIF,
        fontSize: 52, lineHeight: 1.3, opacity: verdictIn}}>{p.footer}</div>
    </Chrome>
  );
};

// ── rows ────────────────────────────────────────────────────────────────────
const Row: React.FC<{r: z.infer<typeof rowSchema>}> = ({r}) => {
  const o = useIn(r.appearAtS);
  const out = useIn(r.outAtS ?? r.appearAtS);
  return (
    <div style={{display: 'flex', alignItems: 'baseline', fontFamily: MONO, fontSize: 56,
      padding: '14px 24px', borderRadius: 12, marginBottom: 10, opacity: o * (r.dim ? 0.35 : 1),
      transform: `translateX(${(1 - o) * -30}px)`,
      background: r.highlight ? 'rgba(217,119,87,0.14)' : 'transparent',
      outline: r.highlight ? `4px solid ${ACC}` : 'none'}}>
      <div style={{width: 560, fontWeight: 700}}>{r.input}</div>
      <div style={{width: 110, fontSize: 88, lineHeight: 0.6, color: CLAUDE.INK_SOFT, opacity: out}}>→</div>
      <div style={{opacity: out}}>{r.output}</div>
    </div>
  );
};
const Rows: React.FC<SoftmaxOffsetProps> = (p) => {
  const errIn = useIn(p.errorAtS);
  const footIn = useIn(p.footer ? p.footerAtS : -1);
  const below = 230 + p.rows.length * 104 + 50;   // first free y under the rows
  return (
    <Chrome {...p}>
      <div style={{position: 'absolute', top: 230, left: 112, right: 112}}>
        {p.rows.map((r, i) => <Row key={i} r={r} />)}
      </div>
      {p.errorInput && (
        <div style={{position: 'absolute', top: p.footer ? below + 110 : below, left: 96, right: 96, opacity: errIn,
          fontFamily: MONO, fontSize: 44, background: '#2A2720', color: CLAUDE.PAGE,
          borderRadius: 14, padding: '22px 30px'}}>
          <div style={{color: '#C9C4B2'}}>{p.errorInput}</div>
          <div style={{color: '#F2A58A', fontWeight: 700}}>{p.errorOutput}</div>
        </div>
      )}
      {p.footer && (
        <div style={{position: 'absolute', top: below, left: 96, right: 96,
          fontFamily: SERIF, fontSize: 50, fontWeight: 700, opacity: footIn}}>{p.footer}</div>
      )}
    </Chrome>
  );
};

// ── card (plain, high-contrast boundary statement) ───────────────────────────
const CardLine: React.FC<{l: {text: string; atS: number; strong: boolean}; first: boolean; light: boolean; bodyScale: number}> = ({l, first, light, bodyScale}) => {
  const o = useIn(l.atS);
  return <div style={{fontFamily: first ? SANS : SERIF, fontSize: first ? 84 : Math.round(66 * bodyScale),
    fontWeight: first || l.strong ? 700 : 400, letterSpacing: first ? 5 : 0, lineHeight: 1.2,
    color: first ? (light ? ACC : '#F2A58A') : (light ? CLAUDE.INK : CLAUDE.PAGE),
    marginBottom: first ? 90 : 52, opacity: o,
    transform: `translateY(${(1 - o) * 12}px)`}}>{l.text}</div>;
};
const Card: React.FC<SoftmaxOffsetProps> = (p) => (
  <AbsoluteFill style={{background: p.light ? CLAUDE.PAGE : '#1F1D18', padding: '0 130px', justifyContent: 'center'}}>
    {p.lines.map((l, i) => <CardLine key={i} l={l} first={i === 0} light={p.light} bodyScale={p.bodyScale} />)}
  </AbsoluteFill>
);

export const SoftmaxOffset: React.FC<SoftmaxOffsetProps> = (p) => {
  switch (p.mode) {
    case 'pipeline': return <Pipeline {...p} />;
    case 'table': return <Table {...p} />;
    case 'shift': return <Shift {...p} />;
    case 'compare': return <Compare {...p} />;
    case 'cancel': return <Cancel {...p} />;
    case 'intermediates': return <Intermediates {...p} />;
    case 'rows': return <Rows {...p} />;
    case 'card': return <Card {...p} />;
  }
};
