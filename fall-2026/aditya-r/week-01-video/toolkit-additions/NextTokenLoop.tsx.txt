/**
 * NextTokenLoop.tsx — reel-local Remotion components for
 * claude-liam-next-token-loop ("A Chatbot Is a Loop.").
 *
 * Palette: the claude fidelity tokens — cream PAGE, warm INK, terracotta SPARK
 * as the ONE accent per beat. All at 1920x1080, registered in Root.tsx under
 * the folder "NextTokenLoop".
 *
 * SOURCE CARE: every probability these components show comes from a recorded
 * local run of SmolLM2-135M-Instruct (greedy decoding) stored in the reel's
 * evidence/ folder. Props carry the numbers; no component hardcodes a value,
 * so a wrong figure is a beat-sheet fix, not a component fix.
 *
 * TIMING: `durationSeconds` = the beat's measured audio (calculateMetadata);
 * every `cue*` prop is a second offset taken from mp3/words.json, so reveals
 * land on the spoken word. Pure functions of frame — no timers, no randomness.
 *
 * LAYOUT: all essential content sits inside SAFE (x 96–1824, y 54–1026).
 */
import React from 'react';
import {AbsoluteFill, Img, staticFile, useCurrentFrame, useVideoConfig, spring, interpolate} from 'remotion';
import {z} from 'zod';
import {CLAUDE, CLAUDE_FONT} from './tokens/claude';
import {SAFE} from './tokens/layout';

const SERIF = CLAUDE_FONT.serif;
const MONO = CLAUDE_FONT.mono;
const SANS = CLAUDE_FONT.ui;
/** pale terracotta plate: ink text on it clears WCAG 4.5:1; the accent stays on the border */
const PLATE = '#F7E1D7';
const cl = (v: number, a = 0, b = 1) => Math.min(b, Math.max(a, v));
const SPRING = {damping: 28, stiffness: 110, mass: 0.9};

/** seconds -> 0..1 spring that starts at `at` seconds */
const useIn = (at: number, delayFrames = 0) => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  return spring({frame: frame - Math.round(at * fps) - delayFrames, fps, config: SPRING});
};
/** linear 0..1 ramp between two times in seconds */
const ramp = (t: number, a: number, b: number) => cl((t - a) / Math.max(0.001, b - a));

/** Percent label that never rounds a probability below 1 up to "100%". */
const pctLabel = (p: number) => {
  const pct = p * 100;
  if (pct < 10) return `${pct.toFixed(1)}%`;
  if (Math.round(pct) >= 100 && p < 1) return `${pct.toFixed(1)}%`;
  return `${Math.round(pct)}%`;
};

/** Visible form of a token: leading space shown as a thin gap, specials as-is. */
const tokenLabel = (tok: string) => (tok.startsWith(' ') ? '␣' + tok.slice(1) : tok);

// ── Shared chrome ────────────────────────────────────────────────────────────

/** SPARK-LINE LAW: spark + one short serif line, top-left inside SAFE. */
const SparkLine: React.FC<{line: string}> = ({line}) => {

  return (

      alignItems: 'baseline', gap: 16, opacity: t}}>
      <span style={{fontFamily: SERIF, fontSize: 60, color: CLAUDE.SPARK, lineHeight: 1}}>✻</span>
      <span style={{fontFamily: SERIF, fontSize: 56, color: CLAUDE.INK_SOFT, lineHeight: 1}}>{line}</span>
    </div>
  );
};

/** LOGO LAW: the @NikBearBrown mark, small + low-opacity, lower-right inside SAFE. */
const LogoBug: React.FC = () => (
  <Img src={staticFile('ntl/nbb-logo.svg')} style={{position: 'absolute',
    right: 1920 - SAFE.r, bottom: 1080 - SAFE.b - 18, width: 165, opacity: 0.32}} />
);

/** Small provenance line, bottom-left inside SAFE. */
const Caption: React.FC<{text: string; at?: number}> = ({text, at = 0.3}) => {
  const t = useIn(at);
  return (
    <div style={{position: 'absolute', left: SAFE.x, bottom: 1080 - SAFE.b + 6,
      maxWidth: SAFE.w - 220, fontFamily: SANS, fontSize: 30, color: CLAUDE.INK_SOFT,
      opacity: t * 0.95, lineHeight: 1.25}}>{text}</div>
  );
};

/** Evidence badge, top-right inside SAFE. The kind is read from the text:
 *  "Constructed…" → dashed outline · "Real Claude response…" → filled ink · anything else → solid outline
 *  (recorded / intervention / computed). One visual vocabulary for evidence across the film. */
const Badge: React.FC<{text: string}> = ({text}) => {
  const t = useIn(0.15);
  const constructed = /^constructed/i.test(text);
  const claude = /^real claude/i.test(text);
  return (
    <div style={{position: 'absolute', right: 1920 - SAFE.r, top: SAFE.y + 8, maxWidth: 1180, whiteSpace: 'nowrap',
      fontFamily: SANS, fontSize: 30, fontWeight: 600, letterSpacing: 0.5, lineHeight: 1.2,
      color: claude ? CLAUDE.CARD : CLAUDE.INK, background: claude ? CLAUDE.INK : CLAUDE.CARD,
      border: `2.5px ${constructed ? 'dashed' : 'solid'} ${CLAUDE.INK}`, borderRadius: 999,
      padding: '9px 22px', opacity: t}}>{text}</div>
  );
};

const Stage: React.FC<{spark: string; caption?: string; badge?: string; children: React.ReactNode}> =
  ({spark, caption, badge, children}) => (
    <AbsoluteFill style={{background: CLAUDE.PAGE, overflow: 'hidden'}}>
      <SparkLine line={spark} />
      {badge ? <Badge text={badge} /> : null}
      {children}
      {caption ? <Caption text={caption} /> : null}
      <LogoBug />
    </AbsoluteFill>
  );

const candidate = z.object({token: z.string(), p: z.number().min(0).max(1)});

// ===========================================================================
// NtlLoop — B02 framework: text so far → model → probabilities → pick & append → back
// ===========================================================================
export const ntlLoopSchema = z.object({
  sparkLine: z.string().default('One step, repeated.'),
  badge: z.string().default(''),
  durationSeconds: z.number().positive().default(6.7),
  context: z.string().default('…What is the capital of France?'),
  candidates: z.array(candidate).default([
    {token: 'The', p: 0.8312}, {token: 'Paris', p: 0.0415}, {token: 'As', p: 0.0183},
  ]),
  cueIn: z.number().default(0.1),
  cueProb: z.number().default(2.25),
  cuePick: z.number().default(4.38),
  cueAppend: z.number().default(5.0),
  cueAgain: z.number().default(5.5),
  caption: z.string().default(''),
});
export type NtlLoopProps = z.infer<typeof ntlLoopSchema>;

export const NtlLoop: React.FC<NtlLoopProps> = ({sparkLine, badge, context, candidates,
  cueIn, cueProb, cuePick, cueAppend, cueAgain, caption}) => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  const t = frame / fps;

  // four stations on a rectangle that fills SAFE below the spark line
  const W = 600, H = 250;
  const L = SAFE.x + 20, R = SAFE.r - 20 - W;
  const TOP = 190, BOT = 690;
  const st = [
    {x: L, y: TOP, title: 'Text so far', at: cueIn},
    {x: R, y: TOP, title: 'The model', at: cueIn + 0.6},
    {x: R, y: BOT, title: 'Next-token probabilities', at: cueProb},
    {x: L, y: BOT, title: 'Pick one · append it', at: cuePick},
  ];
  const cx = (i: number) => st[i].x + W / 2;
  const cy = (i: number) => st[i].y + H / 2;
  // edges: TL→TR (top), TR→BR (right), BR→BL (bottom), BL→TL (left)
  const edges = [
    {x1: L + W, y1: cy(0), x2: R, y2: cy(1), at: cueIn + 0.3},
    {x1: cx(1), y1: TOP + H, x2: cx(2), y2: BOT, at: cueProb - 0.4},
    {x1: R, y1: cy(2), x2: L + W, y2: cy(3), at: cuePick - 0.3},
    {x1: cx(3), y1: BOT, x2: cx(0), y2: TOP + H, at: cueAppend},
  ];
  const chosen = candidates[0];
  const appended = t >= cueAppend + 0.45;
  const maxP = Math.max(...candidates.map((c) => c.p));
  // the chip flies BL → TL on "append"
  const fly = interpolate(t, [cueAppend, cueAppend + 0.45], [0, 1], {extrapolateLeft: 'clamp', extrapolateRight: 'clamp'});
  const againIn = useIn(cueAgain);

  return (
    <Stage spark={sparkLine} caption={caption} badge={badge}>
      <svg width={1920} height={1080} style={{position: 'absolute', left: 0, top: 0}}>
        <defs>
          <marker id="ntl-ah" markerWidth="12" markerHeight="12" refX="10" refY="6" orient="auto">
            <path d="M0,0 L12,6 L0,12 Z" fill={CLAUDE.INK_SOFT} />
          </marker>
        </defs>
        {edges.map((e, i) => {
          const p = ramp(t, e.at, e.at + 0.5);
          return (
            <line key={i} x1={e.x1} y1={e.y1} x2={e.x1 + (e.x2 - e.x1) * p} y2={e.y1 + (e.y2 - e.y1) * p}
              stroke={CLAUDE.INK_SOFT} strokeWidth={4} opacity={p > 0 ? 1 : 0}
              markerEnd={p > 0.98 ? 'url(#ntl-ah)' : undefined} />
          );
        })}
      </svg>

      {st.map((s, i) => {
        const a = cl(spring({frame: frame - Math.round(s.at * fps), fps, config: SPRING}));
        return (
          <div key={i} style={{position: 'absolute', left: s.x, top: s.y, width: W, height: H,
            background: CLAUDE.CARD, border: `2px solid ${CLAUDE.BORDER}`, borderRadius: 20,
            boxSizing: 'border-box', padding: '26px 32px', opacity: a,
            transform: `translateY(${(1 - a) * 18}px)`, display: 'flex', flexDirection: 'column', gap: 18}}>
            <div style={{fontFamily: SERIF, fontSize: 54, color: CLAUDE.INK, lineHeight: 1.05}}>{s.title}</div>
            {i === 0 && (
              <div style={{fontFamily: MONO, fontSize: 34, color: CLAUDE.INK, lineHeight: 1.3}}>
                {context}
                {appended ? <span style={{color: CLAUDE.INK, fontWeight: 700, borderBottom: `5px solid ${CLAUDE.SPARK}`}}>{' ' + chosen.token}</span> : null}
                <span style={{color: CLAUDE.INK_SOFT}}>▍</span>
              </div>
            )}
            {i === 1 && (
              <div style={{fontFamily: SANS, fontSize: 32, color: CLAUDE.INK_SOFT, lineHeight: 1.3}}>
                reads all of it, once per pass
              </div>
            )}
            {i === 2 && (
              <div style={{display: 'flex', flexDirection: 'column', gap: 12}}>
                {candidates.map((c, k) => {
                  const g = ramp(t, cueProb + 0.15 + k * 0.12, cueProb + 0.75 + k * 0.12);
                  const isChosen = k === 0 && t >= cuePick;
                  return (
                    <div key={k} style={{display: 'flex', alignItems: 'center', gap: 16}}>
                      <span style={{fontFamily: MONO, fontSize: 30, width: 110, color: CLAUDE.INK}}>{c.token}</span>
                      <div style={{height: 22, borderRadius: 11, width: 380 * (c.p / maxP) * g + 6,
                        background: isChosen ? CLAUDE.SPARK : CLAUDE.GHOST}} />
                    </div>
                  );
                })}
              </div>
            )}
            {i === 3 && t >= cueAppend + 0.45 && (
              <div style={{fontFamily: SANS, fontSize: 32, color: CLAUDE.INK_SOFT}}>
                picked the top token: <span style={{fontFamily: MONO, color: CLAUDE.INK}}>{chosen.token}</span>
              </div>
            )}
            {i === 3 && t >= cuePick && fly < 1 && (
              <div style={{alignSelf: 'flex-start', fontFamily: MONO, fontSize: 40, color: CLAUDE.INK,
                background: PLATE, border: `3px solid ${CLAUDE.SPARK}`, borderRadius: 12, padding: '8px 22px',
                transform: `translateY(${-fly * (BOT - TOP)}px)`, opacity: 1 - fly * 0.6}}>
                {chosen.token}
              </div>
            )}
          </div>
        );
      })}

      {/* centre: the pass counter — the loop goes around again */}
      <div style={{position: 'absolute', left: 0, right: 0, top: 470, textAlign: 'center',
        fontFamily: SERIF, fontSize: 64, color: CLAUDE.INK, opacity: againIn,
        transform: `scale(${0.9 + 0.1 * againIn})`}}>
        ↻ pass 2
      </div>
    </Stage>
  );
};

// NtlTemplate — B03: the chat, as the one document the model continues
// ===========================================================================
export const ntlTemplateSchema = z.object({
  sparkLine: z.string().default('One document.'),
  badge: z.string().default(''),
  durationSeconds: z.number().positive().default(5.5),
  lines: z.array(z.object({
    special: z.string().default(''),
    text: z.string(),
    kind: z.enum(['role', 'body', 'question']).default('body'),
    end: z.string().default(''),
  })).default([]),
  cueDoc: z.number().default(1.8),
  cueQuestion: z.number().default(2.5),
  cueBlank: z.number().default(4.4),
  blankLabel: z.string().default('the loop writes here'),
  caption: z.string().default(''),
});
export type NtlTemplateProps = z.infer<typeof ntlTemplateSchema>;

const SpecialTok: React.FC<{tok: string}> = ({tok}) => (
  <span style={{fontFamily: MONO, fontSize: 30, color: CLAUDE.INK_SOFT, background: CLAUDE.PILL,
    borderRadius: 8, padding: '2px 10px', margin: '0 6px'}}>{tok}</span>
);

export const NtlTemplate: React.FC<NtlTemplateProps> = ({sparkLine, badge, lines, cueDoc, cueQuestion,
  cueBlank, blankLabel, caption}) => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  const t = frame / fps;
  const card = useIn(0.05);
  const qIn = useIn(cueQuestion);
  const blankIn = useIn(cueBlank);
  const blink = Math.floor(frame / 12) % 2 === 0;
  const perLine = Math.max(0.2, (cueDoc - 0.2) / Math.max(1, lines.length));

  return (
    <Stage spark={sparkLine} caption={caption} badge={badge}>
      <div style={{position: 'absolute', left: SAFE.x, top: 150, width: SAFE.w, height: 760,
        background: CLAUDE.CARD, border: `2px solid ${CLAUDE.BORDER}`, borderRadius: 24,
        boxSizing: 'border-box', padding: '48px 56px', opacity: card,
        display: 'flex', flexDirection: 'column', justifyContent: 'center', gap: 26}}>
        {lines.map((ln, i) => {
          const a = ramp(t, 0.2 + i * perLine, 0.2 + (i + 0.6) * perLine);
          const isQ = ln.kind === 'question';
          return (
            <div key={i} style={{fontFamily: MONO, fontSize: ln.kind === 'role' ? 40 : 34,
              color: CLAUDE.INK, opacity: a, lineHeight: 1.25,
              fontWeight: ln.kind === 'role' ? 700 : 400,
              paddingLeft: isQ ? 18 : 0,
              borderLeft: isQ ? `6px solid rgba(61,57,41,${0.85 * qIn})` : '6px solid transparent'}}>
              {ln.special ? <SpecialTok tok={ln.special} /> : null}
              {ln.text}
              {ln.end ? <SpecialTok tok={ln.end} /> : null}
            </div>
          );
        })}
        <div style={{display: 'flex', alignItems: 'center', gap: 28, opacity: blankIn, height: 60}}>
          <span style={{display: 'inline-block', width: 30, height: 54, borderRadius: 4,
            background: CLAUDE.SPARK, opacity: blink ? 1 : 0.25}} />
          <span style={{fontFamily: SERIF, fontSize: 54, color: CLAUDE.INK}}>{blankLabel}</span>
        </div>
      </div>
    </Stage>
  );
};

// NtlBars — B04 / B06: one pass's top-5 next-token probabilities
// ===========================================================================
export const ntlBarsSchema = z.object({
  sparkLine: z.string().default('How answers start.'),
  badge: z.string().default(''),
  durationSeconds: z.number().positive().default(5),
  heading: z.string().default('Pass 1'),
  context: z.string().default('…<|im_start|>assistant'),
  candidates: z.array(candidate).default([{token: 'The', p: 0.8312}]),
  chosenIndex: z.number().int().default(0),
  ringIndex: z.number().int().default(-1),
  cueGrow: z.number().default(0.8),
  cueRing: z.number().default(3.3),
  caption: z.string().default(''),
});
export type NtlBarsProps = z.infer<typeof ntlBarsSchema>;

export const NtlBars: React.FC<NtlBarsProps> = ({sparkLine, badge, heading, context, candidates,
  chosenIndex, ringIndex, cueGrow, cueRing, caption}) => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  const t = frame / fps;
  const head = useIn(0.05);
  const ringIn = useIn(cueRing);
  const blink = Math.floor(frame / 12) % 2 === 0;

  const LABEL_W = 330, PCT_W = 170;
  const RAIL = SAFE.w - LABEL_W - PCT_W - 60;
  const ROW_TOP = 400, ROW_H = 112;

  return (
    <Stage spark={sparkLine} caption={caption} badge={badge}>
      <div style={{position: 'absolute', left: SAFE.x, top: 150, width: SAFE.w, opacity: head,
        fontFamily: SERIF, fontSize: 64, color: CLAUDE.INK, lineHeight: 1.1}}>{heading}</div>
      <div style={{position: 'absolute', left: SAFE.x, top: 258, maxWidth: SAFE.w, opacity: head,
        fontFamily: MONO, fontSize: 36, color: CLAUDE.INK, background: CLAUDE.CARD,
        border: `2px solid ${CLAUDE.BORDER}`, borderRadius: 14, padding: '14px 26px'}}>
        {context}<span style={{color: CLAUDE.INK_SOFT, opacity: blink ? 1 : 0.2}}> ▍</span>
      </div>

      {candidates.map((c, i) => {
        const g = ramp(t, cueGrow + i * 0.1, cueGrow + 0.7 + i * 0.1);
        const isChosen = i === chosenIndex;
        const y = ROW_TOP + i * ROW_H;
        const pctText = pctLabel(c.p);
        return (
          <div key={i} style={{position: 'absolute', left: SAFE.x, top: y, width: SAFE.w,
            height: ROW_H - 22, display: 'flex', alignItems: 'center', gap: 30}}>
            <div style={{width: LABEL_W, textAlign: 'right', fontFamily: MONO, fontSize: 42,
              color: CLAUDE.INK, fontWeight: isChosen ? 700 : 400}}>{tokenLabel(c.token)}</div>
            <div style={{position: 'relative', width: RAIL, height: 26, borderRadius: 13,
              background: CLAUDE.FOOTER}}>
              <div style={{position: 'absolute', left: 0, top: 0, height: 26, borderRadius: 13,
                width: Math.max(10, RAIL * c.p * g),
                background: isChosen ? CLAUDE.SPARK : CLAUDE.INK_SOFT, opacity: isChosen ? 1 : 0.55}} />
              {i === ringIndex && (
                <div style={{position: 'absolute', left: -14, top: -16, width: RAIL * 0.2,
                  height: 58, borderRadius: 29, border: `4px solid ${CLAUDE.INK}`, opacity: ringIn}} />
              )}
            </div>
            <div style={{width: PCT_W, fontFamily: SERIF, fontSize: 48, color: CLAUDE.INK,
              opacity: g, fontWeight: isChosen ? 700 : 400}}>{pctText}</div>
          </div>
        );
      })}
    </Stage>
  );
};

// ===========================================================================
// NtlUnroll — B05: the loop unrolled, one row per pass
// ===========================================================================
export const ntlUnrollSchema = z.object({
  sparkLine: z.string().default('Append. Predict. Repeat.'),
  badge: z.string().default(''),
  durationSeconds: z.number().positive().default(4.5),
  startContext: z.number().int().default(37),
  steps: z.array(candidate).default([]),
  cueFirst: z.number().default(0.15),
  cueLast: z.number().default(3.3),
  stopToken: z.string().default('<|im_end|>'),
  caption: z.string().default(''),
});
export type NtlUnrollProps = z.infer<typeof ntlUnrollSchema>;

export const NtlUnroll: React.FC<NtlUnrollProps> = ({sparkLine, badge, startContext, steps, cueFirst,
  cueLast, stopToken, caption}) => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  const t = frame / fps;
  const n = Math.max(1, steps.length);
  const at = (i: number) => cueFirst + (i / Math.max(1, n - 1)) * (cueLast - cueFirst);
  const shown = steps.filter((_, i) => t >= at(i));
  const sentence = shown.filter((s) => s.token !== stopToken).map((s) => s.token).join('');
  const stopped = shown.some((s) => s.token === stopToken);

  const COLS = [120, 300, 380, 700];   // pass · context · token · probability
  const ROW_TOP = 330, ROW_H = 72;
  const BAR = 520;

  return (
    <Stage spark={sparkLine} caption={caption} badge={badge}>
      {/* the sentence, assembling */}
      <div style={{position: 'absolute', left: SAFE.x, top: 150, width: SAFE.w, fontFamily: SERIF,
        fontSize: 76, color: CLAUDE.INK, lineHeight: 1.1, whiteSpace: 'pre'}}>
        {sentence}
        {!stopped && <span style={{color: CLAUDE.INK_SOFT}}>▍</span>}
      </div>
      {/* header row */}
      <div style={{position: 'absolute', left: SAFE.x, top: ROW_TOP - 58, display: 'flex',
        fontFamily: SANS, fontSize: 30, color: CLAUDE.INK_SOFT, letterSpacing: 2}}>
        <div style={{width: COLS[0]}}>PASS</div>
        <div style={{width: COLS[1]}}>TOKENS IN</div>
        <div style={{width: COLS[2]}}>APPENDED</div>
        <div style={{width: COLS[3]}}>ITS PROBABILITY</div>
      </div>
      {steps.map((s, i) => {
        const a = cl(spring({frame: frame - Math.round(at(i) * fps), fps, config: SPRING}));
        const isStop = s.token === stopToken;
        return (
          <div key={i} style={{position: 'absolute', left: SAFE.x, top: ROW_TOP + i * ROW_H,
            height: ROW_H - 12, display: 'flex', alignItems: 'center', opacity: a,
            transform: `translateX(${(1 - a) * 30}px)`, borderTop: `1px solid ${CLAUDE.BORDER}`}}>   {/* slide in from the right: never crosses SAFE.x */}
            <div style={{width: COLS[0], fontFamily: SERIF, fontSize: 40, color: CLAUDE.INK}}>{i + 1}</div>
            <div style={{width: COLS[1], fontFamily: MONO, fontSize: 34, color: CLAUDE.INK_SOFT}}>{startContext + i}</div>
            <div style={{width: COLS[2]}}>
              <span style={{fontFamily: MONO, fontSize: 36, color: CLAUDE.INK, fontWeight: isStop ? 700 : 400,
                background: isStop ? PLATE : CLAUDE.PILL, borderRadius: 10,
                padding: '4px 14px'}}>{tokenLabel(s.token)}</span>
            </div>
            <div style={{width: COLS[3], display: 'flex', alignItems: 'center', gap: 22}}>
              <div style={{width: BAR * s.p * a, height: 26, borderRadius: 13,
                background: CLAUDE.INK_SOFT, opacity: 0.55}} />
              <span style={{fontFamily: SERIF, fontSize: 38, color: CLAUDE.INK}}>{pctLabel(s.p)}</span>
              {isStop && <span style={{fontFamily: SERIF, fontSize: 38, color: CLAUDE.INK}}>→ end of turn</span>}
            </div>
          </div>
        );
      })}
    </Stage>
  );
};

                {c.quote ? <div style={{fontFamily: SERIF, fontSize: 38, color: CLAUDE.INK, fontStyle: 'italic', marginTop: 12,
                  borderLeft: `5px solid ${CLAUDE.INK_SOFT}`, paddingLeft: 20}}>{c.quote}
                  <span style={{fontFamily: SANS, fontStyle: 'normal', fontSize: 30, color: CLAUDE.INK_SOFT}}> — {c.source}</span></div> : null}
              </div>
            </div>
          );
        })}
      </div>
    </Stage>
  );
};

// ===========================================================================
// NtlClaudeReply — a REAL Claude response the author received, verbatim and dated.
// The component only lays it out; the text must come from the author's own session.
// ===========================================================================
export const ntlClaudeReplySchema = z.object({
  sparkLine: z.string().default('Confidence is text too.'),
  badge: z.string().default(''),
  durationSeconds: z.number().positive().default(15),
  prompt: z.string().default(''),
  reply: z.string().default(''),
  highlight: z.string().default(''),
  meta: z.string().default(''),
  annotation: z.string().default(''),
  cueReply: z.number().default(1.0),
  cueHighlight: z.number().default(6.0),
  cueNote: z.number().default(9.0),
  caption: z.string().default(''),
});
export type NtlClaudeReplyProps = z.infer<typeof ntlClaudeReplySchema>;

export const NtlClaudeReply: React.FC<NtlClaudeReplyProps> = ({sparkLine, badge, prompt, reply, highlight, meta,
  annotation, cueReply, cueHighlight, cueNote, caption}) => {
  const frame = useCurrentFrame();
  const {fps} = useVideoConfig();
  const t = frame / fps;
  const replyIn = useIn(cueReply);
  const hl = ramp(t, cueHighlight, cueHighlight + 0.6);
  const noteIn = useIn(cueNote);
  const size = reply.length > 420 ? 30 : reply.length > 260 ? 34 : 40;
  const at = highlight ? reply.indexOf(highlight) : -1;
  const parts = at >= 0 ? [reply.slice(0, at), reply.slice(at, at + highlight.length), reply.slice(at + highlight.length)] : [reply, '', ''];
  return (
    <Stage spark={sparkLine} caption={caption} badge={badge}>
      <div style={{position: 'absolute', left: SAFE.x, top: 150, width: SAFE.w}}>
        <div style={{fontFamily: SANS, fontSize: 30, color: CLAUDE.INK_SOFT}}>{meta}</div>
        <div style={{marginTop: 14, display: 'inline-block', maxWidth: SAFE.w, fontFamily: SANS, fontSize: 32, color: CLAUDE.INK,
          background: CLAUDE.PILL, borderRadius: 18, padding: '16px 26px'}}>{prompt}</div>
        <div style={{marginTop: 26, background: CLAUDE.CARD, border: `2px solid ${CLAUDE.BORDER}`, borderRadius: 20,
          padding: '28px 34px', fontFamily: SERIF, fontSize: size, lineHeight: 1.35, color: CLAUDE.INK,
          whiteSpace: 'pre-wrap', opacity: replyIn}}>
          {parts[0]}
          <span style={{backgroundImage: `linear-gradient(${CLAUDE.SPARK}, ${CLAUDE.SPARK})`, backgroundRepeat: 'no-repeat',
            backgroundPosition: '0 95%', backgroundSize: `${hl * 100}% 5px`}}>{parts[1]}</span>
          {parts[2]}
        </div>
        <div style={{marginTop: 26, fontFamily: SERIF, fontSize: 42, color: CLAUDE.INK, opacity: noteIn,
          borderLeft: `6px solid ${CLAUDE.INK}`, paddingLeft: 24}}>{annotation}</div>
      </div>
    </Stage>
  );
};
