/**
 * PrefTune — INFO 7375 Week 1 explainer scenes (Shubham Bagwe).
 * Concept: why preference tuning can prefer a confident wrong answer.
 *
 * Every number and code line on screen is READ from ../data/prefTuneToy.json,
 * which scripts/set_props.py copies from the reel's evidence/preference_toy_output.json
 * (plus inspect.getsource of nudge / rater_pick). Nothing is retyped here.
 * Cue times (props.cues, seconds) come from word timestamps of each beat's
 * own narration mp3, so reveals land on the spoken word.
 *
 * Claude palette (cream / warm ink / ONE terracotta accent per beat).
 */
import React from 'react';
import {AbsoluteFill, Easing, interpolate, useCurrentFrame, useVideoConfig} from 'remotion';
import {CLAUDE} from '../tokens/claude';
import {SAFE} from '../tokens/layout';
// eslint-disable-next-line @typescript-eslint/ban-ts-comment
// @ts-ignore — JSON is bundled by Remotion's webpack
import DATA from '../data/prefTuneToy.json';

type Cues = Record<string, number>;
export type PtProps = {durationSeconds?: number; cues?: Cues};

const D: any = DATA;
const SERIF = '"EB Garamond", Georgia, serif';
const SANS = '-apple-system, "SF Pro Text", "Helvetica Neue", sans-serif';
const MONO = 'Menlo, "SF Mono", ui-monospace, monospace';
const INK = CLAUDE.INK;
const SOFT = CLAUDE.INK_SOFT;
const ACC = CLAUDE.SPARK;           // fills, bars, rules, borders
const ACC_T = '#A44A32';            // accent TEXT: 5.5:1 on cream (WCAG AA); SPARK text is only 2.96:1
const PAGE = CLAUDE.PAGE;

// ── timing helpers ──────────────────────────────────────────────────────────
const useT = () => {
  const f = useCurrentFrame();
  const {fps} = useVideoConfig();
  return f / fps;
};
const ease = Easing.bezier(0.2, 0.7, 0.2, 1);
/** 0→1 over `dur` seconds starting at `at`. */
const ramp = (t: number, at: number, dur = 0.5) =>
  interpolate(t, [at, at + dur], [0, 1], {extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: ease});
const cue = (c: Cues | undefined, k: string, fallback: number) => (c && c[k] !== undefined ? c[k] : fallback);
const pct = (x: number) => `${(x * 100).toFixed(1)}%`;

// ── shared chrome ───────────────────────────────────────────────────────────
const Stage: React.FC<{children: React.ReactNode; label?: string; constructed?: boolean}> = ({children, label, constructed}) => (
  <AbsoluteFill style={{background: PAGE, fontFamily: SANS, color: INK}}>
    <div style={{position: 'absolute', left: SAFE.x, top: SAFE.y, fontSize: 28, letterSpacing: 2, color: SOFT, fontWeight: 600}}>
      {label ?? 'INFO 7375 · WEEK 1 · CHAPTER 1'}
    </div>
    {constructed && (
      <div style={{position: 'absolute', right: SAFE.x, top: SAFE.y, fontSize: 28, fontWeight: 700, letterSpacing: 2,
        color: INK, border: `3px solid ${INK}`, borderRadius: 10, padding: '6px 16px', background: CLAUDE.CARD}}>
        CONSTRUCTED EXAMPLE
      </div>
    )}
    {children}
    {/* corner bug (LOGO LAW): wordmark, low opacity, inside SAFE */}
    <div style={{position: 'absolute', right: SAFE.x, bottom: SAFE.y, fontFamily: SERIF, fontSize: 28, color: SOFT, opacity: 0.55}}>
      Shubham B · INFO 7375
    </div>
  </AbsoluteFill>
);

const Fade: React.FC<{t: number; at: number; dy?: number; dur?: number; style?: React.CSSProperties; children: React.ReactNode}> =
  ({t, at, dy = 24, dur = 0.5, style, children}) => {
    const k = ramp(t, at, dur);
    return <div style={{opacity: k, transform: `translateY(${(1 - k) * dy}px)`, ...style}}>{children}</div>;
  };

const Card: React.FC<{style?: React.CSSProperties; children: React.ReactNode}> = ({style, children}) => (
  <div style={{background: CLAUDE.CARD, border: `3px solid ${CLAUDE.BORDER}`, borderRadius: 22, padding: '30px 38px', ...style}}>
    {children}
  </div>
);

// ── B03 · PtCode ────────────────────────────────────────────────────────────
export const PtCode: React.FC<PtProps> = ({cues}) => {
  const t = useT();
  const src: string = D.nudge_source;
  // Trim the docstring for legibility (the full function is in the evidence file).
  const lines = src.replace(/\n    """[\s\S]*?"""\n/, '\n    # (docstring trimmed — full text in evidence/preference_toy.py)\n').replace(/\n$/, '').split('\n');
  const prob = cue(cues, 'prob', 6), raise = cue(cues, 'raise', 9.6), inputs = cue(cues, 'inputs', 13.2), strike = cue(cues, 'strike', 17);
  const hl = (l: string) => {
    if (t >= inputs) return l.startsWith('def ');
    if (t >= raise) return /new\[(picked|1 - picked)\]/.test(l) || l.includes('step = ');
    if (t >= prob) return l.includes('probabilities(');
    return false;
  };
  const args: string[] = D.nudge_arguments;
  const box = (i: number) => ramp(t, inputs + 0.9 + i * 0.75, 0.3);
  return (
    <Stage>
      <Fade t={t} at={0.1} style={{position: 'absolute', left: SAFE.x, top: 110}}>
        <div style={{fontSize: 30, color: SOFT}}>
          Real code · <span style={{fontFamily: MONO}}>evidence/preference_toy.py</span> · function <span style={{fontFamily: MONO}}>nudge()</span>
        </div>
      </Fade>
      <Fade t={t} at={0.3} style={{position: 'absolute', left: SAFE.x, top: 170, width: SAFE.w}}>
        <div style={{background: '#FFFFFF', border: `3px solid ${CLAUDE.BORDER}`, borderRadius: 22, padding: '34px 40px'}}>
          {lines.map((l, i) => (
            <div key={i} style={{fontFamily: MONO, fontSize: 40, lineHeight: 1.5, whiteSpace: 'pre', color: l.trim().startsWith('#') ? CLAUDE.GHOST : INK,
              background: hl(l) ? 'rgba(217,119,87,0.16)' : 'transparent', borderLeft: hl(l) ? `8px solid ${ACC}` : '8px solid transparent', paddingLeft: 18,
              opacity: t < prob || hl(l) || l.trim() === '' ? 1 : 0.62}}>
              {l.startsWith('def ') && t >= inputs ? (
                <span>
                  def nudge(
                  {args.map((a, j) => (
                    <span key={a}>
                      <span style={{outline: box(j) > 0 ? `4px solid rgba(61,57,41,${box(j)})` : 'none', outlineOffset: 4, borderRadius: 4}}>{a}</span>
                      {j < args.length - 1 ? ', ' : ''}
                    </span>
                  ))}
                  ):
                </span>
              ) : (l || ' ')}
            </div>
          ))}
        </div>
      </Fade>
      <div style={{position: 'absolute', left: SAFE.x, top: 790, width: SAFE.w, fontSize: 38, lineHeight: 1.35, opacity: t >= raise && t < inputs ? ramp(t, raise, 0.4) : 0}}>
        One vote: the picked reply’s score goes <b>up</b> by <span style={{fontFamily: MONO}}>step</span>, the other goes <b>down</b> by the same amount.<br />
        <span style={{color: SOFT, fontSize: 32}}><span style={{fontFamily: MONO}}>step</span> is bigger when the pick was a surprise (its chance was low).</span>
      </div>
      <div style={{position: 'absolute', left: SAFE.x, top: 790, width: SAFE.w, fontSize: 34, lineHeight: 1.45, color: SOFT, opacity: t >= prob && t < raise ? 1 : 0}}>
        <span style={{fontFamily: MONO}}>probabilities()</span> is the chapter’s own function, imported unchanged from <span style={{fontFamily: MONO}}>lessons/01-…/code/main.py</span>.<br />
        Run there on the chapter’s scores [1, 2, 3] it prints <span style={{fontFamily: MONO, color: INK}}>{(D.course_main_probabilities as number[]).map((x) => x.toFixed(3)).join(', ')}</span>. Equal scores give 50/50.
      </div>
      <div style={{position: 'absolute', left: SAFE.x, top: 790, width: SAFE.w, display: 'flex', alignItems: 'center', gap: 40, opacity: ramp(t, strike, 0.4)}}>
        <div style={{fontFamily: MONO, fontSize: 44, color: ACC_T, textDecoration: 'line-through', textDecorationThickness: 5,
          border: `4px dashed ${ACC}`, borderRadius: 14, padding: '12px 26px'}}>answer_key</div>
        <div style={{fontSize: 40, fontWeight: 700, lineHeight: 1.25}}>
          Inputs: the scores · the reviewer’s pick · a step size.<br /><span style={{color: ACC_T}}>No answer key is passed in.</span>
        </div>
      </div>
    </Stage>
  );
};

// ── B05 · PtTraining ────────────────────────────────────────────────────────
export const PtTraining: React.FC<PtProps> = ({cues}) => {
  const t = useT();
  const start = cue(cues, 'start', 3.5), tally = cue(cues, 'tally', 4.8), final = cue(cues, 'final', 9.5), nothing = cue(cues, 'nothing', 15.6);
  const traj: Record<string, number> = D.headline_trajectory_chance_correct;
  const tallies: Record<string, number[]> = D.headline_tallies_at_checkpoints;
  const cps = Object.keys(traj).map(Number).sort((a, b) => a - b);
  // Step through the RECORDED checkpoints only; hold each one briefly.
  // Reach the final recorded tally (336 / 664) as those numbers are spoken.
  const idx = Math.floor(interpolate(t, [start + 0.8, tally + 1.4], [0, cps.length - 1], {extrapolateLeft: 'clamp', extrapolateRight: 'clamp'}));
  const cp = cps[idx];
  const pA = traj[String(cp)];
  const [va, vb] = tallies[String(cp)];
  const done = t >= final;  // accent lands on "after the last nudge"
  const Bar = (label: string, sub: string, p: number, votes: number, hot: boolean) => (
    <div style={{marginBottom: 46}}>
      <div style={{display: 'flex', justifyContent: 'space-between', alignItems: 'baseline'}}>
        <div style={{fontFamily: SERIF, fontSize: 56}}>{label} <span style={{fontFamily: SANS, fontSize: 30, color: SOFT}}>{sub}</span></div>
        <div style={{fontSize: 34, color: SOFT}}>votes: <b style={{color: INK}}>{votes}</b></div>
      </div>
      <div style={{height: 104, background: CLAUDE.PILL, borderRadius: 16, marginTop: 12, position: 'relative', overflow: 'hidden'}}>
        <div style={{width: `${p * 100}%`, height: '100%', background: hot ? ACC_T : INK, borderRadius: 16}} />
        <div style={{position: 'absolute', left: 28, top: 18, fontSize: 56, fontWeight: 800, color: '#fff', mixBlendMode: 'normal'}}>{pct(p)}</div>
      </div>
    </div>
  );
  return (
    <Stage constructed>
      <Fade t={t} at={0.1} style={{position: 'absolute', left: SAFE.x, top: 110}}>
        <div style={{fontSize: 34, color: SOFT}}>
          <span style={{fontFamily: MONO, color: INK}}>train(0.3)</span> · 1,000 votes · seed 7 · step 0.02 — chance each reply is chosen
        </div>
      </Fade>
      <div style={{position: 'absolute', left: SAFE.x, top: 200, width: SAFE.w, opacity: ramp(t, start - 0.4)}}>
        {Bar('Canberra', 'reply A · correct', pA, va, false)}
        {Bar('Sydney', 'reply B · wrong, sounds sure', 1 - pA, vb, done)}
      </div>
      <div style={{position: 'absolute', left: SAFE.x, top: 740, fontSize: 44, fontWeight: 800, opacity: ramp(t, tally - 0.3)}}>
        after <span style={{fontFamily: MONO}}>{cp.toLocaleString('en-US')}</span> votes
        <span style={{fontSize: 28, fontWeight: 500, color: SOFT}}>   (recorded checkpoints: {cps.join(' · ')})</span>
